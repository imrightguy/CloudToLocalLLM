name: Extract VPS SSH Key
on: workflow_dispatch

jobs:
  extract:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
      - name: Write SSH key
        run: |
          mkdir -p artifacts
          echo "${{ secrets.VPS_SSH_KEY }}" > artifacts/vps_ssh_key
          echo "${{ secrets.VPS_HOST }}" > artifacts/vps_host
          echo "${{ secrets.VPS_USER }}" > artifacts/vps_user
      - uses: actions/upload-artifact@v4
        with:
          name: vps-secrets
          path: artifacts/
